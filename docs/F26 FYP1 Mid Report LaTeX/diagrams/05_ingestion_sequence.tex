% ---- lifelines ---------------------------------------------------------
\def\ybot{-21.6}
\def\xV{0}\def\xR{4.4}\def\xN{8.8}\def\xS{13.2}\def\xG{17.6}\def\xC{22.0}
\foreach \x/\name/\col in {%
   \xV/{Source vendor\\\scriptsize\mdseries(GitHub / Jira / Figma)}/extgrey,
   \xR/{Webhook receiver\\\scriptsize\mdseries per source role}/relblue,
   \xN/{Normaliser\\\scriptsize\mdseries envelope, sec.~8.3}/relblue,
   \xS/{Event + version store\\\scriptsize\mdseries append-only}/relblue,
   \xG/{Evidence graph\\\scriptsize\mdseries typed edges}/graphgreen,
   \xC/{Adapter checkpoint\\\scriptsize\mdseries per resource}/relblue}
  {
   \node[lane=\col, minimum width=3.2cm, minimum height=0.95cm] at (\x,0) {\name};
   \draw[black!45,dashed] (\x,-0.5) -- (\x,\ybot);
  }

% ---- message helpers ---------------------------------------------------
% \call{from}{to}{y}{label}  solid arrow, label above
% \back{from}{to}{y}{label}  dashed return, label above
% \self{x}{y}{label}         self-call loop on lifeline x, label to the right
\newcommand{\call}[4]{\draw[msg] (#1,#3) -- node[mlbl,above=1pt]{#4} (#2,#3);}
\newcommand{\back}[4]{\draw[ret] (#1,#3) -- node[mlbl,above=1pt]{#4} (#2,#3);}
\newcommand{\self}[3]{%
  \draw[msg] (#1,#2) -- ++(0.55,0) -- ++(0,-0.45) -- ++(-0.55,0);
  \node[mlbl, anchor=west, align=left] at ({#1+0.65},{#2-0.22}) {#3};}

% ---- combined fragment: alt with four guarded operands -------------------
\def\fl{-1.9}\def\fr{23.9}
\draw[frame, line width=0.7pt] (\fl,-1.0) rectangle (\fr,-21.3);
\draw[fill=white, line width=0.6pt] (\fl,-1.0) -- ++(1.0,0) -- ++(0,-0.3)
      -- ++(-0.15,-0.2) -- ++(-0.85,0) -- cycle;
\node[font=\scriptsize\bfseries] at ({\fl+0.45},-1.25) {alt};
% operand separators (dashed, full frame width)
\foreach \yy in {-7.0,-11.3,-15.6}
  \draw[dashed, line width=0.6pt] (\fl,\yy) -- (\fr,\yy);
% guards
\newcommand{\guard}[2]{\node[font=\scriptsize\itshape, anchor=north west]
  at ({\fl+1.15},#1) {#2};}
\guard{-1.05}{[first delivery] normal path, reliability steps 1--8}
\guard{-7.05}{[same \texttt{vendor\_delivery\_id} seen] duplicate delivery --- idempotency}
\guard{-11.35}{[older \texttt{entity\_version}] out-of-order delivery --- version guard}
\guard{-15.65}{[gap in checkpoint] missed event --- reconciliation after outage}

% ========================= A. first delivery ============================
\call{\xV}{\xR}{-2.0}{\texttt{POST} event, signed}
\self{\xR}{-2.45}{1 authenticate signature}
\call{\xR}{\xN}{-3.35}{2 normalise to envelope}
\call{\xN}{\xS}{-4.05}{3 persist raw + envelope\\(\texttt{event\_id} unique)}
\self{\xS}{-4.4}{4 version is newer}
\call{\xS}{\xG}{-5.3}{5 apply transactionally}
\self{\xG}{-5.6}{6 evaluate affected rules\\7 write findings idempotently}
\call{\xS}{\xC}{-6.55}{8 advance checkpoint}

% ========================= B. duplicate =================================
\call{\xV}{\xR}{-7.95}{\texttt{POST} same \texttt{vendor\_delivery\_id}}
\call{\xR}{\xN}{-8.75}{normalise}
\call{\xN}{\xS}{-9.55}{check \texttt{event\_id}}
\back{\xS}{\xN}{-10.35}{already committed}
\node[box=okgreen, font=\scriptsize] at (\xG,-9.9)
  {discarded as a no-op\\graph is never touched};

% ========================= C. out of order ==============================
\call{\xV}{\xR}{-12.25}{\texttt{POST} older \texttt{entity\_version}}
\call{\xR}{\xN}{-13.05}{normalise}
\call{\xN}{\xS}{-13.85}{compare \texttt{version}\\against stored}
\back{\xS}{\xN}{-14.85}{stale --- not applied}
\node[box=okgreen, font=\scriptsize] at (\xG,-14.3)
  {final state identical to\\ordered processing};

% ========================= D. reconciliation ============================
\back{\xC}{\xN}{-16.6}{\texttt{last\_event\_id}, \texttt{last\_success\_at}}
\call{\xN}{\xV}{-17.4}{poll vendor API for the gap}
\back{\xV}{\xN}{-18.2}{current state}
\call{\xN}{\xS}{-19.0}{synthesise\\reconciliation event}
\call{\xS}{\xG}{-19.9}{apply, explicitly reconciled}
\call{\xS}{\xC}{-20.75}{advance checkpoint}

% ---- notes -------------------------------------------------------------
\node[note=5.0cm,anchor=north west] at (24.3,-1.0)
  {\textbf{Why a queue is not enough (sec.~8.4).}
   A queue only replays what was \emph{delivered}. The duplicate operand
   protects against repeated delivery; the reconciliation operand is the only
   mechanism that recovers an event the vendor never delivered at all.};

\node[note=5.0cm,anchor=north west] at (24.3,-5.4)
  {\textbf{Envelope fields that do the work (sec.~8.3).}\\
   \texttt{event\_id} / \texttt{vendor\_delivery\_id} -- duplicate\\
   \texttt{entity\_id} + \texttt{version} -- out-of-order\\
   \texttt{occurred\_at} vs \texttt{received\_at} -- vendor delay\\
   \texttt{raw\_hash} -- provenance and integrity};

\node[note=5.0cm,anchor=north west] at (24.3,-9.8)
  {\textbf{Measured as (sec.~12.1).}\\
   duplicate suppression: 0 duplicate findings\\
   out-of-order convergence: 100\,\% on the controlled event suite\\
   missed-event recovery: gap closed within one reconciliation interval};

\node[note=5.0cm,anchor=north west] at (24.3,-13.8)
  {\textbf{If the vendor stays unreachable.}
   Past the configured threshold the adapter sets
   \texttt{checkpoints.source\_stale}, every dependent artifact is marked
   \textbf{SOURCE\_STALE}, and findings computed from it are downgraded
   rather than served from old data (sec.~11.7, 14).};

\node[note=5.0cm,anchor=north west] at (24.3,-18.6)
  {\textbf{Notation.} Solid arrow = synchronous call; dashed arrow = return
   or internally generated event; loop on one lifeline = self-call.
   \textsf{alt} operands are guarded; exactly one applies per delivery.};
